# Hattorio Tiling Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A verified, dependency-free Lua library that lazily generates the "hat" aperiodic tiling in exact integer arithmetic, plus the Python pipeline that derives and validates its data.

**Architecture:** A Python pipeline ports hatviz's combinatorial substitution into exact arithmetic over Q(phi, sqrt3), validates the resulting tiling (overlap, gaps, reflected density, coefficient bounds), and emits Lua data files containing only integer literals. Six pure Lua modules consume that data to answer one question — which hats overlap this box — with no floating point in the composition path and no Factorio dependency anywhere.

**Tech Stack:** Python 3.11+ (`fractions`, stdlib only), Lua 5.2, busted, luassert, lua-quickcheck, luacheck.

**Spec:** `docs/superpowers/specs/2026-09-19-hattorio-design.md`

## Global Constraints

- **Lua 5.2.1** — no integer subtype, all numbers are doubles, exact-integer ceiling **2^53**. No `goto`, no integer division operator.
- **No floating point in the composition path.** Ring arithmetic is integers only. `to_float` exists solely for consumers.
- **No `math.sin`, `math.cos`, or `^`/`pow` anywhere in shipped Lua.** Not platform-deterministic.
- **No division in the Lua ring.** Totality depends on it.
- **Ring and transform values are immutable.** Every operation returns a new table; none mutates an argument. The sole exception is the descent accumulator in `hat/tiling.lua`.
- **`hat/` modules have zero Factorio dependencies.** They must load under plain `lua5.2`.
- **Ring element** is `{a, b, c, d}` meaning `(a + b*phi) + (c + d*phi)*sqrt(3)`, all integers.
- **Denominator invariant:** substitution rule transforms have denominator 1; only level-0 hat placements may carry one. Asserted by the pipeline.
- **Attribution:** hatviz is BSD-3-Clause (isohedral/hatviz). Its notice ships in `THIRD_PARTY.md`.
- Generated files carry a `-- GENERATED by tools/emit_lua.py -- DO NOT EDIT` header.

```python
"""Exact arithmetic over Q(phi, sqrt3), degree 4 over Q."""
from __future__ import annotations
from fractions import Fraction as F
from math import lcm, sqrt

_PHI = (1 + sqrt(5)) / 2
_SQRT3 = sqrt(3)


class Q5:
    """p + q*phi, with phi^2 = phi + 1."""
    __slots__ = ("p", "q")

    def __init__(self, p, q=0):
        self.p = F(p)
        self.q = F(q)

    def __add__(self, o): return Q5(self.p + o.p, self.q + o.q)
    def __sub__(self, o): return Q5(self.p - o.p, self.q - o.q)
    def __neg__(self):    return Q5(-self.p, -self.q)

    def __mul__(self, o):
        # (p+q phi)(r+s phi) = (pr + qs) + (ps + qr + qs) phi
        return Q5(self.p * o.p + self.q * o.q,
                  self.p * o.q + self.q * o.p + self.q * o.q)

    def inverse(self):
        # norm = p^2 + pq - q^2 ; conjugate is (p + q) - q phi
        n = self.p * self.p + self.p * self.q - self.q * self.q
        if n == 0:
            raise ZeroDivisionError("Q5 inverse of zero")
        return Q5((self.p + self.q) / n, -self.q / n)

    def __truediv__(self, o): return self * o.inverse()
    def __eq__(self, o): return isinstance(o, Q5) and self.p == o.p and self.q == o.q
    def __hash__(self): return hash((self.p, self.q))
    def __repr__(self): return f"Q5({self.p}, {self.q})"
    def to_float(self): return float(self.p) + float(self.q) * _PHI
    def is_zero(self): return self.p == 0 and self.q == 0


Q5.ZERO = Q5(0, 0)
Q5.ONE = Q5(1, 0)
Q5.PHI = Q5(0, 1)


class K:
    """x + y*sqrt(3), with x, y in Q(phi)."""
    __slots__ = ("x", "y")

    def __init__(self, x: Q5, y: Q5 = None):
        self.x = x
        self.y = y if y is not None else Q5.ZERO

    @staticmethod
    def from_int(n): return K(Q5(n, 0), Q5.ZERO)

    @staticmethod
    def half(): return K(Q5(F(1, 2), 0), Q5.ZERO)

    def __add__(self, o): return K(self.x + o.x, self.y + o.y)
    def __sub__(self, o): return K(self.x - o.x, self.y - o.y)
    def __neg__(self):    return K(-self.x, -self.y)

    def __mul__(self, o):
        # (x + y r3)(u + v r3) = (xu + 3yv) + (xv + yu) r3
        three = Q5(3, 0)
        return K(self.x * o.x + three * (self.y * o.y),
                 self.x * o.y + self.y * o.x)

    def inverse(self):
        # (x + y r3)^-1 = (x - y r3) / (x^2 - 3 y^2)
        three = Q5(3, 0)
        n = self.x * self.x - three * (self.y * self.y)
        if n.is_zero():
            raise ZeroDivisionError("K inverse of zero")
        ni = n.inverse()
        return K(self.x * ni, -self.y * ni)

    def __truediv__(self, o): return self * o.inverse()
    def __eq__(self, o): return isinstance(o, K) and self.x == o.x and self.y == o.y
    def __hash__(self): return hash((self.x, self.y))
    def __repr__(self): return f"K({self.x}, {self.y})"
    def is_zero(self): return self.x.is_zero() and self.y.is_zero()

    def to_float(self):
        return self.x.to_float() + self.y.to_float() * _SQRT3

    def denominator(self) -> int:
        return lcm(self.x.p.denominator, self.x.q.denominator,
                   self.y.p.denominator, self.y.q.denominator)

    def is_integral(self) -> bool:
        return self.denominator() == 1

    def numerators(self):
        """(a, b, c, d) integers such that self == ((a + b phi) + (c + d phi) r3) / denominator()."""
        d = self.denominator()
        return (int(self.x.p * d), int(self.x.q * d),
                int(self.y.p * d), int(self.y.q * d))


K.ZERO = K(Q5.ZERO, Q5.ZERO)
K.ONE = K(Q5.ONE, Q5.ZERO)
K.SQRT3 = K(Q5.ZERO, Q5.ONE)
K.PHI = K(Q5.PHI, Q5.ZERO)
```

```python
"""Exact points and affine transforms over K = Q(phi, sqrt3)."""
from __future__ import annotations
from fractions import Fraction as F
from math import lcm
from tools.field import K, Q5


def hr3() -> K:
    """sqrt(3)/2."""
    return K(Q5.ZERO, Q5(F(1, 2), 0))


class Pt:
    __slots__ = ("x", "y")

    def __init__(self, x: K, y: K):
        self.x, self.y = x, y

    def __add__(self, o): return Pt(self.x + o.x, self.y + o.y)
    def __sub__(self, o): return Pt(self.x - o.x, self.y - o.y)
    def __eq__(self, o): return isinstance(o, Pt) and self.x == o.x and self.y == o.y
    def __hash__(self): return hash((self.x, self.y))
    def __repr__(self): return f"Pt({self.x}, {self.y})"
    def to_float(self): return (self.x.to_float(), self.y.to_float())


class Xf:
    """(a b c / d e f): maps (x,y) -> (ax+by+c, dx+ey+f)."""
    __slots__ = ("m",)

    def __init__(self, m):
        assert len(m) == 6
        self.m = tuple(m)

    @staticmethod
    def identity():
        z, o = K.ZERO, K.ONE
        return Xf([o, z, z, z, o, z])

    @staticmethod
    def translate(p: Pt):
        z, o = K.ZERO, K.ONE
        return Xf([o, z, p.x, z, o, p.y])

    @staticmethod
    def scale(k: K):
        z = K.ZERO
        return Xf([k, z, z, z, k, z])

    @staticmethod
    def rot60():
        # cos60 = 1/2, sin60 = sqrt(3)/2
        h, s, z = K.half(), hr3(), K.ZERO
        return Xf([h, -s, z, s, h, z])

    @staticmethod
    def flip_y():
        z, o = K.ZERO, K.ONE
        return Xf([o, z, z, z, -o, z])

    def mul(self, o: "Xf") -> "Xf":
        a, b, c, d, e, f = self.m
        A, B, C, D, E, F_ = o.m
        return Xf([a * A + b * D, a * B + b * E, a * C + b * F_ + c,
                   d * A + e * D, d * B + e * E, d * C + e * F_ + f])

    def apply(self, p: Pt) -> Pt:
        a, b, c, d, e, f = self.m
        return Pt(a * p.x + b * p.y + c, d * p.x + e * p.y + f)

    def det(self) -> K:
        a, b, _, d, e, _ = self.m
        return a * e - b * d

    def is_reflected(self) -> bool:
        return self.det().to_float() < 0

    def denominator(self) -> int:
        return lcm(*[k.denominator() for k in self.m])

    def numerators(self):
        d = self.denominator()
        out = []
        for k in self.m:
            scaled = k * K.from_int(d)
            out.append(scaled.numerators())
        return tuple(out)

    def __eq__(self, o): return isinstance(o, Xf) and self.m == o.m
    def __repr__(self): return f"Xf({[str(k) for k in self.m]})"


def match_two(p1: Pt, p2: Pt, q1: Pt, q2: Pt) -> Xf:
    """The orientation-preserving similarity carrying p1->q1 and p2->q2."""
    u = p2 - p1
    v = q2 - q1
    # complex division v/u with i -> (x,y): s = v * conj(u) / |u|^2
    den = u.x * u.x + u.y * u.y
    if den.is_zero():
        raise ZeroDivisionError("match_two on a degenerate segment")
    inv = den.inverse()
    sx = (v.x * u.x + v.y * u.y) * inv
    sy = (v.y * u.x - v.x * u.y) * inv
    linear = Xf([sx, -sy, K.ZERO, sy, sx, K.ZERO])
    return Xf.translate(q1).mul(linear).mul(Xf.translate(Pt(-p1.x, -p1.y)))


def _p(xh, yh) -> Pt:
    """Vertex at (xh, yh*sqrt(3)) where xh, yh are halves-of-integers."""
    return Pt(K(Q5(F(xh)), Q5.ZERO), K(Q5.ZERO, Q5(F(yh))))


HAT_OUTLINE = [
    _p(F(0), F(0)),        _p(F(-3, 2), F(-1, 2)), _p(F(0), F(-1)),
    _p(F(3, 2), F(-1, 2)), _p(F(3), F(-1)),        _p(F(9, 2), F(-1, 2)),
    _p(F(3), F(0)),        _p(F(3), F(1)),         _p(F(3, 2), F(3, 2)),
    _p(F(0), F(1)),        _p(F(0), F(2)),         _p(F(-3, 2), F(5, 2)),
    _p(F(-3), F(2)),       _p(F(-3), F(1)),
]
```

```python
"""Level-0 metatiles, ported from hatviz (BSD-3-Clause). See THIRD_PARTY.md."""
from __future__ import annotations
from fractions import Fraction as F
from tools.field import K, Q5
from tools.geometry import Pt, Xf, match_two, HAT_OUTLINE, hr3


class Child:
    __slots__ = ("xf", "geom")

    def __init__(self, xf: Xf, geom):
        self.xf, self.geom = xf, geom


class MetaTile:
    __slots__ = ("label", "outline", "children")

    def __init__(self, label, outline, children=None):
        self.label = label
        self.outline = list(outline)
        self.children = list(children or [])

    def add(self, xf: Xf, geom):
        self.children.append(Child(xf, geom))


def _p(xh, yh) -> Pt:
    return Pt(K(Q5(F(xh)), Q5.ZERO), K(Q5.ZERO, Q5(F(yh))))


def hat_polygon(xf: Xf):
    return [xf.apply(v) for v in HAT_OUTLINE]


def _half_scale() -> Xf:
    return Xf.scale(K.half())
```

```python
from __future__ import annotations
from tools.field import K, Q5
from tools.geometry import Pt, Xf, match_two, hr3
from tools.hatviz_port import MetaTile, Child, level0

RULES = [
    ("H",),
    (0, 0, "P", 2), (1, 0, "H", 2), (2, 0, "P", 2), (3, 0, "H", 2),
    (4, 4, "P", 2), (0, 4, "F", 3), (2, 4, "F", 3),
    (4, 1, 3, 2, "F", 0),
    (8, 3, "H", 0), (9, 2, "P", 0), (10, 2, "H", 0), (11, 4, "P", 2),
    (12, 0, "H", 2), (13, 0, "F", 3), (14, 2, "F", 1), (15, 3, "H", 4),
    (8, 2, "F", 1), (17, 3, "H", 0), (18, 2, "P", 0), (19, 2, "H", 2),
    (20, 4, "F", 3), (20, 0, "P", 2), (22, 0, "H", 2), (23, 4, "F", 3),
    (23, 0, "F", 3), (16, 0, "P", 2),
    (9, 4, 0, 2, "T", 2),
    (4, 0, "F", 3),
]


def rotate_about(centre: Pt, sixths: int) -> Xf:
    r = Xf.identity()
    step = Xf.rot60()
    for _ in range(sixths % 6):
        r = r.mul(step)
    back = Xf.translate(Pt(-centre.x, -centre.y))
    return Xf.translate(centre).mul(r).mul(back)


def line_intersect(a1: Pt, a2: Pt, b1: Pt, b2: Pt) -> Pt:
    d1 = a2 - a1
    d2 = b2 - b1
    den = d1.x * d2.y - d1.y * d2.x
    if den.is_zero():
        raise ZeroDivisionError("parallel lines in line_intersect")
    w = a1 - b1
    t = (d2.x * w.y - d2.y * w.x) * den.inverse()
    return Pt(a1.x + d1.x * t, a1.y + d1.y * t)


def _eval_child(mt: MetaTile, idx: int, vert: int) -> Pt:
    ch = mt.children[idx]
    poly = ch.geom.outline if isinstance(ch.geom, MetaTile) else None
    assert poly is not None, "eval_child on a hat leaf"
    return ch.xf.apply(poly[vert])


def construct_patch(shapes) -> MetaTile:
    out = MetaTile("patch", [])
    for r in RULES:
        if len(r) == 1:
            out.add(Xf.identity(), shapes[r[0]])
        elif len(r) == 4:
            src = out.children[r[0]]
            poly = src.geom.outline
            P = src.xf.apply(poly[(r[1] + 1) % len(poly)])
            Q = src.xf.apply(poly[r[1]])
            nsh = shapes[r[2]]
            npoly = nsh.outline
            out.add(match_two(npoly[r[3]], npoly[(r[3] + 1) % len(npoly)], P, Q), nsh)
        else:
            chP = out.children[r[0]]
            chQ = out.children[r[2]]
            P = chQ.xf.apply(chQ.geom.outline[r[3]])
            Q = chP.xf.apply(chP.geom.outline[r[1]])
            nsh = shapes[r[4]]
            npoly = nsh.outline
            out.add(match_two(npoly[r[5]], npoly[(r[5] + 1) % len(npoly)], P, Q), nsh)
    return out


def _recentre(mt: MetaTile) -> MetaTile:
    c = mt.outline[0]
    back = Xf.translate(Pt(-c.x, -c.y))
    mt.outline = [Pt(p.x - c.x, p.y - c.y) for p in mt.outline]
    mt.children = [Child(back.mul(ch.xf), ch.geom) for ch in mt.children]
    return mt


def construct_metatiles(patch: MetaTile):
    bps1 = _eval_child(patch, 8, 2)
    bps2 = _eval_child(patch, 21, 2)
    rbps = rotate_about(bps1, 4).apply(bps2)          # -2pi/3 == +4 sixths
    p72 = _eval_child(patch, 7, 2)
    p252 = _eval_child(patch, 25, 2)
    c62 = _eval_child(patch, 6, 2)

    llc = line_intersect(bps1, rbps, c62, p72)
    w = c62 - llc

    rot_m60 = Xf.rot60().mul(Xf.rot60()).mul(Xf.rot60()) \
                       .mul(Xf.rot60()).mul(Xf.rot60())   # +5 sixths == -60 deg

    H_out = [llc, bps1]
    w = Pt(*_lin(rot_m60, w))
    H_out.append(Pt(H_out[1].x + w.x, H_out[1].y + w.y))
    H_out.append(_eval_child(patch, 14, 2))
    w = Pt(*_lin(rot_m60, w))
    H_out.append(Pt(H_out[3].x - w.x, H_out[3].y - w.y))
    H_out.append(c62)
    newH = MetaTile("H", H_out,
                    [patch.children[i] for i in (0, 9, 16, 27, 26, 6, 1, 8, 10, 15)])

    P_out = [p72, Pt(p72.x + (bps1.x - llc.x), p72.y + (bps1.y - llc.y)), bps1, llc]
    newP = MetaTile("P", P_out, [patch.children[i] for i in (7, 2, 3, 4, 28)])

    F_out = [bps2, _eval_child(patch, 24, 2), _eval_child(patch, 25, 0), p252,
             Pt(p252.x + (llc.x - bps1.x), p252.y + (llc.y - bps1.y))]
    newF = MetaTile("F", F_out, [patch.children[i] for i in (21, 20, 22, 23, 24, 25)])

    AAA = H_out[2]
    BBB = Pt(H_out[1].x + (H_out[4].x - H_out[5].x),
             H_out[1].y + (H_out[4].y - H_out[5].y))
    CCC = rotate_about(BBB, 5).apply(AAA)
    newT = MetaTile("T", [BBB, CCC, AAA], [patch.children[11]])

    return {"H": _recentre(newH), "T": _recentre(newT),
            "P": _recentre(newP), "F": _recentre(newF)}


def _lin(xf: Xf, p: Pt):
    """Apply only the linear part of xf to p."""
    a, b, _, d, e, _ = xf.m
    return (a * p.x + b * p.y, d * p.x + e * p.y)


def levels(n: int):
    out = [level0()]
    for _ in range(n):
        out.append(construct_metatiles(construct_patch(out[-1])))
    return out
```

```python
"""Validation of generated hat patches. Catches sign errors before any Lua exists."""
from __future__ import annotations
import math
from tools.geometry import Xf
from tools.hatviz_port import MetaTile, hat_polygon
from tools.substitution import levels

PHI = (1 + math.sqrt(5)) / 2


def flatten(mt: MetaTile, acc: Xf = None):
    acc = acc or Xf.identity()
    out = []
    for ch in mt.children:
        xf = acc.mul(ch.xf)
        if isinstance(ch.geom, str):
            out.append((xf, xf.is_reflected()))
        else:
            out.extend(flatten(ch.geom, xf))
    return out


def _poly_f(xf: Xf):
    return [p.to_float() for p in hat_polygon(xf)]


def _area(poly):
    s = 0.0
    for i in range(len(poly)):
        x1, y1 = poly[i]
        x2, y2 = poly[(i + 1) % len(poly)]
        s += x1 * y2 - x2 * y1
    return abs(s) / 2


def _inside(poly, x, y):
    ins = False
    n = len(poly)
    j = n - 1
    for i in range(n):
        ax, ay = poly[i]
        bx, by = poly[j]
        if (ay > y) != (by > y):
            if x < ax + (y - ay) / (by - ay) * (bx - ax):
                ins = not ins
        j = i
    return ins


def _centroid(poly):
    return (sum(p[0] for p in poly) / len(poly), sum(p[1] for p in poly) / len(poly))


def check_no_overlaps(hats, samples=200):
    """Sample interior points of each hat; none may fall inside another."""
    errs = []
    polys = [_poly_f(x) for x, _ in hats]
    import random
    rng = random.Random(12345)
    for i, pi in enumerate(polys):
        xs = [p[0] for p in pi]
        ys = [p[1] for p in pi]
        hits = 0
        tries = 0
        while hits < min(samples, 20) and tries < samples * 20:
            tries += 1
            x = rng.uniform(min(xs), max(xs))
            y = rng.uniform(min(ys), max(ys))
            if not _inside(pi, x, y):
                continue
            hits += 1
            for j, pj in enumerate(polys):
                if i != j and _inside(pj, x, y):
                    errs.append(f"hats {i} and {j} overlap near ({x:.4f}, {y:.4f})")
                    return errs
    return errs


def check_no_gaps(hats, resolution=400):
    """Total hat area must equal the area of their union's bounding sample."""
    polys = [_poly_f(x) for x, _ in hats]
    total = sum(_area(p) for p in polys)
    xs = [p[0] for poly in polys for p in poly]
    ys = [p[1] for poly in polys for p in poly]
    # sample well inside the patch to avoid boundary effects
    cx, cy = (min(xs) + max(xs)) / 2, (min(ys) + max(ys)) / 2
    r = min(max(xs) - min(xs), max(ys) - min(ys)) / 6
    covered = 0
    tested = 0
    step = 2 * r / resolution
    y = cy - r
    while y < cy + r:
        x = cx - r
        while x < cx + r:
            tested += 1
            if any(_inside(p, x, y) for p in polys):
                covered += 1
            x += step
        y += step
    if tested and covered / tested < 0.999:
        return [f"gaps found: only {100*covered/tested:.3f}% of the interior is covered"]
    if total <= 0:
        return ["degenerate patch: zero total area"]
    return []


def reflected_density(hats):
    return sum(1 for _, r in hats if r) / len(hats)


def max_coefficient(hats):
    worst = 0
    for xf, _ in hats:
        for group in xf.numerators():
            for n in group:
                worst = max(worst, abs(int(n)))
        worst = max(worst, xf.denominator())
    return worst


def verify_all(depth: int):
    errs = []
    hats = flatten(levels(depth)[depth]["H"])
    if not hats:
        return ["no hats produced"]
    errs += check_no_overlaps(hats)
    errs += check_no_gaps(hats)
    d = reflected_density(hats)
    if not (0.05 < d < 0.22):
        errs.append(f"reflected density {d:.4f} outside plausible range")
    mc = max_coefficient(hats)
    if mc >= 2 ** 53:
        errs.append(f"coefficient {mc} exceeds 2^53")
    return errs
```

```python
import pathlib, math
from tools.field import K
from tools.geometry import HAT_OUTLINE
from tools.substitution import levels
from tools.hatviz_port import MetaTile

def _ints(k: K, den: int):
    scaled = k * K.from_int(den)
    a, b, c, d = scaled.numerators()
    assert scaled.is_integral(), f"non-integral after scaling: {k}"
    return (a, b, c, d)


def _xf_ints(xf, den):
    out = []

def emit_rules(path):

    lines = [HEADER, "return {\n  max_level = %d,\n  levels = {\n" % MAX_LEVEL]
    for i, shapes in enumerate(lv):
        lines.append("    [%d] = {\n" % i)
        for name, mt in shapes.items():
            lines.append("      %s = {\n" % name)
            for ch in mt.children:
                den = ch.xf.denominator()
                label = ch.geom if isinstance(ch.geom, str) else ch.geom.label
                nums = ", ".join(str(n) for n in _xf_ints(ch.xf, den))
                lines.append('        { shape = "%s", den = %d, xf = { %s } },\n'
                             % (label, den, nums))
            lines.append("      },\n")
        lines.append("    },\n")
    lines.append("  },\n  radius = {\n")
    for i, shapes in enumerate(lv):
        rad = ", ".join("%s = %.9g" % (n, _outline_radius(mt) * 1.05)
```

```lua
-- Exact arithmetic in Z[phi][sqrt3].
-- An element is { a, b, c, d } meaning (a + b*phi) + (c + d*phi)*sqrt(3).
-- Integers only. No division: every operation is total.
-- Values are immutable; every function returns a new table.

local E = {}

local PHI = (1 + math.sqrt(5)) / 2
local SQRT3 = math.sqrt(3)

function E.new(a, b, c, d)
  return { a or 0, b or 0, c or 0, d or 0 }
end

E.ZERO  = E.new(0, 0, 0, 0)
E.ONE   = E.new(1, 0, 0, 0)
E.PHI   = E.new(0, 1, 0, 0)
E.SQRT3 = E.new(0, 0, 1, 0)

function E.add(x, y)
  return { x[1]+y[1], x[2]+y[2], x[3]+y[3], x[4]+y[4] }
end

function E.sub(x, y)
  return { x[1]-y[1], x[2]-y[2], x[3]-y[3], x[4]-y[4] }
end

function E.neg(x)
  return { -x[1], -x[2], -x[3], -x[4] }
end

-- Q(phi) multiplication: (p+q phi)(r+s phi) = (pr+qs) + (ps+qr+qs) phi
local function q5mul(p, q, r, s)
  return p*r + q*s, p*s + q*r + q*s
end

function E.mul(x, y)
  -- (X + Y sqrt3)(U + V sqrt3) = (XU + 3YV) + (XV + YU) sqrt3
  local xu1, xu2 = q5mul(x[1], x[2], y[1], y[2])
  local yv1, yv2 = q5mul(x[3], x[4], y[3], y[4])
  local xv1, xv2 = q5mul(x[1], x[2], y[3], y[4])
  local yu1, yu2 = q5mul(x[3], x[4], y[1], y[2])
  return { xu1 + 3*yv1, xu2 + 3*yv2, xv1 + yu1, xv2 + yu2 }
end

function E.eq(x, y)
  return x[1] == y[1] and x[2] == y[2] and x[3] == y[3] and x[4] == y[4]
end

function E.to_float(x)
  return (x[1] + x[2]*PHI) + (x[3] + x[4]*PHI)*SQRT3
end

function E.max_abs(x)
  local m = 0
  for i = 1, 4 do
    local v = x[i] < 0 and -x[i] or x[i]
    if v > m then m = v end
  end
  return m
end

return E
```

```lua
-- Affine transforms over the exact ring, with an integer denominator.
-- xf = { m = { a, b, c, d, e, f }, den = n }
-- maps (x, y) -> ((a*x + b*y + c)/n, (d*x + e*y + f)/n)
-- Immutable: every operation returns a new table.

local E = require("hat.exact")

local T = {}

function T.new(m, den)
  return { m = m, den = den or 1 }
end

T.IDENTITY = T.new({ E.ONE, E.ZERO, E.ZERO, E.ZERO, E.ONE, E.ZERO }, 1)

function T.from_ints(ints, den)
  local m = {}
  for i = 1, 6 do
    local o = (i - 1) * 4
    m[i] = E.new(ints[o+1], ints[o+2], ints[o+3], ints[o+4])
  end
  return T.new(m, den or 1)
end

function T.mul(x, y)
  local a,b,c,d,e,f = x.m[1],x.m[2],x.m[3],x.m[4],x.m[5],x.m[6]
  local A,B,C,D,EE,F = y.m[1],y.m[2],y.m[3],y.m[4],y.m[5],y.m[6]
  local cy = E.new(y.den, 0, 0, 0)
  return T.new({
    E.add(E.mul(a,A), E.mul(b,D)),
    E.add(E.mul(a,B), E.mul(b,EE)),
    E.add(E.add(E.mul(a,C), E.mul(b,F)), E.mul(c, cy)),
    E.add(E.mul(d,A), E.mul(e,D)),
    E.add(E.mul(d,B), E.mul(e,EE)),
    E.add(E.add(E.mul(d,C), E.mul(e,F)), E.mul(f, cy)),
  }, x.den * y.den)
end

function T.apply_float(xf, px, py)
  local m, n = xf.m, xf.den
  local a,b,c = E.to_float(m[1]), E.to_float(m[2]), E.to_float(m[3])
  local d,e,f = E.to_float(m[4]), E.to_float(m[5]), E.to_float(m[6])
  return (a*px + b*py + c)/n, (d*px + e*py + f)/n
end

function T.origin_float(xf)
  return E.to_float(xf.m[3])/xf.den, E.to_float(xf.m[6])/xf.den
end

local function det_float(xf)
  local m = xf.m
  local a,b = E.to_float(m[1]), E.to_float(m[2])
  local d,e = E.to_float(m[4]), E.to_float(m[5])
  return (a*e - b*d) / (xf.den * xf.den)
end

function T.is_reflected(xf)
  return det_float(xf) < 0
end

function T.scale_float(xf)
  local dt = det_float(xf)
  return math.sqrt(dt < 0 and -dt or dt)
end

function T.max_coefficient(xf)
  local worst = xf.den
  for i = 1, 6 do
    local v = E.max_abs(xf.m[i])
    if v > worst then worst = v end
  end
  return worst
end

return T
```

```lua
-- Hat outline and world-space geometry. Floats here are leaf queries only:
-- deterministic (+ - * / sqrt only), never part of transform composition.

local E = require("hat.exact")
local T = require("hat.transform")
local DATA = require("data.hat_geometry")

local G = {}

-- Decode the emitted outline: pairs of ring elements over a common denominator.
G.OUTLINE = {}
for i = 1, 14 do
  local xv = DATA.outline[(i-1)*2 + 1]
  local yv = DATA.outline[(i-1)*2 + 2]
  G.OUTLINE[i] = {
    E.to_float(E.new(xv[1], xv[2], xv[3], xv[4])) / DATA.den,
    E.to_float(E.new(yv[1], yv[2], yv[3], yv[4])) / DATA.den,
  }
end

G.RADIUS = 0
for i = 1, 14 do
  local p = G.OUTLINE[i]
  local r = math.sqrt(p[1]*p[1] + p[2]*p[2])
  if r > G.RADIUS then G.RADIUS = r end
end

function G.polygon(xf, unit)
  local out = {}
  for i = 1, 14 do
    local p = G.OUTLINE[i]
    local x, y = T.apply_float(xf, p[1], p[2])
    out[i] = { x = x * unit, y = y * unit }
  end
  return out
end

function G.edges(xf, unit)
  local poly = G.polygon(xf, unit)
  local out = {}
  for i = 1, 14 do
    local a = poly[i]
    local b = poly[i % 14 + 1]
    out[i] = { a.x, a.y, b.x, b.y }
  end
  return out
end

function G.point_in_polygon(px, py, poly)
  local inside = false
  local n = #poly
  local j = n
  for i = 1, n do
    local a, b = poly[i], poly[j]
    if (a.y > py) ~= (b.y > py) then
      local t = (py - a.y) / (b.y - a.y)
      if px < a.x + t * (b.x - a.x) then inside = not inside end
    end
    j = i
  end
  return inside
end

function G.dist_to_segment(px, py, x1, y1, x2, y2)
  local vx, vy = x2 - x1, y2 - y1
  local l2 = vx*vx + vy*vy
  local t = 0
  if l2 > 0 then
    t = ((px - x1)*vx + (py - y1)*vy) / l2
    if t < 0 then t = 0 elseif t > 1 then t = 1 end
  end
  local dx, dy = px - (x1 + t*vx), py - (y1 + t*vy)
  return math.sqrt(dx*dx + dy*dy)
end

return G
```

```lua
-- Stable hat identity: the descent path from root to hat, packed as bytes.
-- Exact by construction. Prefixes are ancestors, so ancestor(key, k) names
-- the level-k supertile. Deliberately NOT string.format on a float, which
-- is %.14g and lossy.

local I = {}

function I.encode(path)
  local parts = {}
  for i = 1, #path do
    local v = path[i]
    assert(v >= 0 and v <= 254, "child index out of range: " .. tostring(v))
    parts[i] = string.char(v + 1)
  end
  return table.concat(parts)
end

function I.decode(key)
  local out = {}
  for i = 1, #key do
    out[i] = key:byte(i) - 1
  end
  return out
end

function I.ancestor(key, level)
  return key:sub(1, level)
end

function I.depth(key)
  return #key
end

return I
```

`hat/tiling.lua`:
```lua
-- Lazy descent over the substitution tree. Emits only hats that could touch
-- the query box; cost is proportional to output, not to tiling size.
--
-- Exact ring arithmetic composes the transforms. Pruning is float-only, which
-- is safe: it uses +,-,*,/ and sqrt, all IEEE-deterministic across platforms.

local T = require("hat.transform")
local G = require("hat.geometry")
local I = require("hat.index")
local RULES = require("data.hat_rules")

local Ti = {}
Ti.__index = Ti

function Ti.new(opts)
  local self = setmetatable({}, Ti)
  self.unit = opts.unit or 1
  self.depth = opts.depth or 3
  self.root_shape = opts.root_shape or "H"
  self.root_xf = opts.root_xf or T.IDENTITY
  assert(self.depth <= RULES.max_level,
         "depth " .. self.depth .. " exceeds generated data (" .. RULES.max_level .. ")")
  return self
end

function Ti:depth_for_radius(r)
  -- radius[d][shape] already encodes the level-d outline's own size, so this
  -- is a direct lookup rather than a power of the inflation factor.
  for d = 0, RULES.max_level do
    if RULES.radius[d][self.root_shape] * self.unit >= r then return d end
  end
  return RULES.max_level
end

local function circle_misses_box(cx, cy, rad, box)
  local dx = box.left - cx
  local dx2 = cx - box.right
  if dx2 > dx then dx = dx2 end
  if dx < 0 then dx = 0 end
  local dy = box.top - cy
  local dy2 = cy - box.bottom
  if dy2 > dy then dy = dy2 end
  if dy < 0 then dy = 0 end
  return dx*dx + dy*dy > rad*rad
end

function Ti:_descend(shape, xf, level, box, path, out)
  local cx, cy = T.origin_float(xf)
  cx, cy = cx * self.unit, cy * self.unit
  local k = T.scale_float(xf) * self.unit
  -- Per level: construct_metatiles grows the outlines, so the level-N radius
  -- already includes the inflation. Multiplying by INFLATE^level here (as the
  -- original scaffold did) over-estimates by ~1e5 at depth 11 and disables
  -- pruning entirely.
  local rad = RULES.radius[level][shape] * k

  if circle_misses_box(cx, cy, rad, box) then return end

  local rules = RULES.levels[level][shape]
  local n = #path

  for i = 1, #rules do
    local r = rules[i]
    local child = T.mul(xf, T.from_ints(r.xf, r.den))
    path[n+1] = i - 1
    if level == 0 then
      local px, py = T.origin_float(child)
      px, py = px * self.unit, py * self.unit
      local hr = G.RADIUS * T.scale_float(child) * self.unit
      if not circle_misses_box(px, py, hr, box) then
        out[#out+1] = {
          xf = child,
          reflected = T.is_reflected(child),
          path = I.encode(path),
        }
      end
    else
      self:_descend(r.shape, child, level - 1, box, path, out)
    end
    path[n+1] = nil
  end
end

function Ti:hats_in_box(box)
  local out = {}
  self:_descend(self.root_shape, self.root_xf, self.depth, box, {}, out)
  return out
end

return Ti
```
